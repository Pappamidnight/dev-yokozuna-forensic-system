# cook.ps1
# Comando para interação entre agentes

param(
    [string]$Message,
    [string]$Agent = "all",
    [switch]$Status,
    [switch]$Reset
)

# ============================================================
# CONFIGURACAO
# ============================================================
$CookDir = "C:\Users\nunom\Desktop\cook"
$AgentsDir = "$CookDir\agents"
$SharedDir = "$CookDir\shared"
$OutputsDir = "$CookDir\outputs"

# Criar estrutura se nao existir
if (-not (Test-Path $CookDir)) {
    New-Item -ItemType Directory -Path $CookDir -Force | Out-Null
    New-Item -ItemType Directory -Path $AgentsDir -Force | Out-Null
    New-Item -ItemType Directory -Path $SharedDir -Force | Out-Null
    New-Item -ItemType Directory -Path $OutputsDir -Force | Out-Null
    
    # Criar estrutura de agentes
    @("codex", "antigravity", "claude", "opencode") | ForEach-Object {
        New-Item -ItemType Directory -Path "$AgentsDir\$_" -Force | Out-Null
        @"
# $_ - Ultimo Output
# Atualizado: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
# Status: Aguardando interacao
"@ | Out-File "$AgentsDir\$_\last_output.md" -Encoding UTF8
    }
    
    # Criar contexto compartilhado
    @"
{
    "last_update": "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')",
    "agents": ["codex", "antigravity", "claude", "opencode"],
    "context": {}
}
"@ | Out-File "$SharedDir\context.json" -Encoding UTF8
    
    Write-Host "Estrutura COOK criada em: $CookDir" -ForegroundColor Green
}

# ============================================================
# FUNCOES
# ============================================================

function Write-AgentOutput {
    param(
        [string]$Agent,
        [string]$Content,
        [string]$Timestamp = (Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
    )
    
    $outputPath = "$AgentsDir\$Agent\last_output.md"
    $header = "# AGENTE: $Agent`n# ATUALIZADO: $Timestamp`n`n"
    $header + $Content | Out-File $outputPath -Encoding UTF8
    
    # Criar output histórico
    $historyDir = "$OutputsDir\$($Timestamp -replace '[-:]', '' -replace ' ', '_')"
    New-Item -ItemType Directory -Path $historyDir -Force | Out-Null
    $Content | Out-File "$historyDir\$Agent.md" -Encoding UTF8
}

function Get-AgentOutput {
    param([string]$Agent)
    
    $outputPath = "$AgentsDir\$Agent\last_output.md"
    if (Test-Path $outputPath) {
        return Get-Content $outputPath -Raw -Encoding UTF8
    }
    return "Agente $Agent nao encontrado"
}

function Update-Context {
    param([string]$Key, [string]$Value)
    
    $contextPath = "$SharedDir\context.json"
    $context = Get-Content $contextPath -Raw -Encoding UTF8 | ConvertFrom-Json
    $context.context | Add-Member -NotePropertyName $Key -NotePropertyValue $Value -Force
    $context.last_update = (Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
    $context | ConvertTo-Json -Depth 10 | Out-File $contextPath -Encoding UTF8
}

function Show-Status {
    Write-Host ""
    Write-Host "STATUS DO COOK" -ForegroundColor Cyan
    Write-Host "="*60
    Write-Host ""
    
    $agents = @("codex", "antigravity", "claude", "opencode")
    foreach ($agent in $agents) {
        $outputPath = "$AgentsDir\$agent\last_output.md"
        if (Test-Path $outputPath) {
            $firstLine = Get-Content $outputPath -First 3
            $status = if ($firstLine -match "Atualizado: (.+)") { $matches[1] } else { "Desconhecido" }
            Write-Host "$agent : $status" -ForegroundColor Gray
        }
    }
    
    Write-Host ""
    Write-Host "Contexto:" -ForegroundColor Yellow
    $context = Get-Content "$SharedDir\context.json" -Raw -Encoding UTF8 | ConvertFrom-Json
    Write-Host "  Ultima atualizacao: $($context.last_update)" -ForegroundColor Gray
    Write-Host "  Chaves: $($context.context.PSObject.Properties.Name -join ', ')" -ForegroundColor Gray
    Write-Host ""
}

function Reset-Cook {
    Write-Host "Resetando COOK..." -ForegroundColor Yellow
    
    @("codex", "antigravity", "claude", "opencode") | ForEach-Object {
        @"
# $_ - Ultimo Output
# Atualizado: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
# Status: Resetado
"@ | Out-File "$AgentsDir\$_\last_output.md" -Encoding UTF8
    }
    
    @"
{
    "last_update": "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')",
    "agents": ["codex", "antigravity", "claude", "opencode"],
    "context": {}
}
"@ | Out-File "$SharedDir\context.json" -Encoding UTF8
    
    Write-Host "COOK resetado" -ForegroundColor Green
}

# ============================================================
# PROCESSAMENTO DE COMANDOS
# ============================================================

if ($Reset) {
    Reset-Cook
    exit 0
}

if ($Status) {
    Show-Status
    exit 0
}

# ============================================================
# COMANDO COOK
# ============================================================

if ($Message) {
    $timestamp = Get-Date -Format 'yyyy-MM-dd HH:mm:ss'
    
    Write-Host ""
    Write-Host "COOK - Processando mensagem" -ForegroundColor Cyan
    Write-Host "="*60
    Write-Host "Mensagem: $Message" -ForegroundColor White
    Write-Host "Agente: $Agent" -ForegroundColor White
    Write-Host ""
    
    # Atualizar contexto
    Update-Context -Key "ultima_mensagem" -Value $Message
    
    # Processar agentes
    $agents = @()
    if ($Agent -eq "all") {
        $agents = @("codex", "antigravity", "claude", "opencode")
    } else {
        $agents = @($Agent)
    }
    
    foreach ($agent in $agents) {
        Write-Host "Processando: $agent" -ForegroundColor Yellow
        
        # Simular output do agente (aqui pode integrar chamadas reais)
        $output = @"
## RESPOSTA DO AGENTE: $agent

**Timestamp:** $timestamp
**Mensagem:** $Message

### Analise:

1. Recebi a mensagem: "$Message"
2. Processando com os recursos disponiveis
3. Integrando com contexto anterior
4. Gerando resposta deterministica

### Resposta:

A mensagem foi processada com sucesso pelo agente $agent.

### Contexto:

- Agente: $agent
- Timestamp: $timestamp
- Estado: OK

### Proxima acao:

Aguardando proximo comando COOK.

---
*Gerado automaticamente pelo sistema COOK*
"@
        
        Write-AgentOutput -Agent $agent -Content $output -Timestamp $timestamp
        Write-Host "  Atualizado: $agent" -ForegroundColor Green
    }
    
    Write-Host ""
    Write-Host "COOK processado com sucesso" -ForegroundColor Green
    Write-Host "Ver: $AgentsDir" -ForegroundColor Gray
    Write-Host ""
} else {
    Show-Status
}

```powershell
@($CommandsDir, $TemplatesDir, $LogsDir) | ForEach-Object {
    if (-not (Test-Path $_)) { New-Item -ItemType Directory -Path $_ -Force | Out-Null }

$CommandRegistry = @{
    # Indexacao
    "index" = @{
        Description = "Indexar ficheiros com hash deterministico"
        Usage = "index -Path <caminho> [-Recursive] [-Output <ficheiro>]"
        Function = { 
            param($Params)
            $path = if ($Params.Path) { $Params.Path } else { "." }
            $recursive = $Params.ContainsKey("Recursive")
            $output = if ($Params.Output) { $Params.Output } else { "C:\Users\nunom\Desktop\index_$(Get-Date -Format 'yyyyMMdd_HHmmss').csv" }
            
            $Files = if ($recursive) {
                Get-ChildItem -Path $path -File -Recurse
            } else {
                Get-ChildItem -Path $path -File
            }
            
            "HASH|CAMINHO|NOME|TAMANHO|EXTENSAO|ULTIMA_MODIFICACAO|DIRETORIO" | Out-File $output -Encoding UTF8
            
            foreach ($File in $Files) {
                $Hash = (Get-FileHash -Path $File.FullName -Algorithm MD5 -ErrorAction SilentlyContinue).Hash
                if (-not $Hash) { $Hash = "NAO_DISPONIVEL" }
                
                "$Hash|$($File.FullName)|$($File.Name)|$($File.Length)|$($File.Extension)|$($File.LastWriteTime)|$($File.Directory)" |
                    Out-File $output -Append -Encoding UTF8
            }
            
            Write-Host "Index criado: $output" -ForegroundColor Green
            Write-Host "Total ficheiros: $($Files.Count)" -ForegroundColor Green
        }
    }

                return
            }

                Write-Host "PROCESSO: $numero" -ForegroundColor Cyan
                Write-Host "="*60
                $results | ForEach-Object { Write-Host "  $_" -ForegroundColor Gray }
                
                $refFile = "C:\Users\nunom\Desktop\processo_${numero}_refs.txt"
                $results | Out-File $refFile -Encoding UTF8
                Write-Host "Referencias salvas em: $refFile" -ForegroundColor Green

            }
        }
    }

        Description = "Gerar relatorio de um processo"

        Function = {
            param($Params)
            $numero = $Params.Target
            $output = if ($Params.Output) { $Params.Output } else { "C:\Users\nunom\Desktop\relatorio_${numero}_$(Get-Date -Format 'yyyyMMdd').md" }
            $basePath = if ($Params.Path) { $Params.Path } else { "C:\Users\nunom\Desktop\Yo_Kozuna_fire\PROCESSOS" }
            
            $pattern = $numero -replace '/', '[/_-]'
            $results = rg -l $pattern $basePath --glob "!*.pdf" --glob "!*.exe" --glob "!*.dll" 2>$null
            
            "RELATORIO DO PROCESSO $numero" | Out-File $output -Encoding UTF8
            "="*60 | Out-File $output -Append -Encoding UTF8
            "Gerado em: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')" | Out-File $output -Append -Encoding UTF8
```

```python
#!/usr/bin/env python3
# indexer.py - Indexador deterministico de ficheiros

import os
import sys
import json
import hashlib
import argparse
from pathlib import Path
from datetime import datetime
from concurrent.futures import ThreadPoolExecutor, as_completed

class DeterministicIndexer:
    def __init__(self, root_path, recursive=True, max_workers=8):
        self.root_path = Path(root_path)
        self.recursive = recursive
        self.max_workers = max_workers
        self.results = []
    
    def hash_file(self, file_path):
        """Calcula hash SHA256 do ficheiro"""
        try:
            sha256 = hashlib.sha256()
            with open(file_path, 'rb') as f:
                for chunk in iter(lambda: f.read(65536), b''):
                    sha256.update(chunk)
            return sha256.hexdigest()[:8]
        except:
            return "NAO_DISPONIVEL"
    
    def scan_file(self, file_path):
        """Scan de um ficheiro"""
        try:
            stat = file_path.stat()
            return {
                'hash': self.hash_file(file_path),
                'path': str(file_path),
                'name': file_path.name,
                'size': stat.st_size,
                'extension': file_path.suffix,
                'modified': datetime.fromtimestamp(stat.st_mtime).isoformat(),
                'parent': str(file_path.parent)
            }
        except Exception as e:
            return None
    
    def scan_directory(self):
        """Scan do diretorio"""
        pattern = '**/*' if self.recursive else '*'
        files = list(self.root_path.glob(pattern))
        files = [f for f in files if f.is_file()]
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {executor.submit(self.scan_file, f): f for f in files}
            for future in as_completed(futures):
                result = future.result()
                if result:
                    self.results.append(result)
        
        return self.results
    
    def to_csv(self, output_path):
        """Exportar para CSV"""
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write("HASH|CAMINHO|NOME|TAMANHO|EXTENSAO|ULTIMA_MODIFICACAO|DIRETORIO\n")
            for r in self.results:
                f.write(f"{r['hash']}|{r['path']}|{r['name']}|{r['size']}|{r['extension']}|{r['modified']}|{r['parent']}\n")
    
    def to_json(self, output_path):
        """Exportar para JSON"""
        with open(output_path, 'w', encoding='utf-8') as f:
            json.dump({
                'scan_time': datetime.now().isoformat(),
                'path': str(self.root_path),
                'total_files': len(self.results),
                'files': self.results
            }, f, ensure_ascii=False, indent=2)

def main():
    parser = argparse.ArgumentParser(description='Indexador deterministico')
    parser.add_argument('path', help='Caminho para indexar')
    parser.add_argument('-o', '--output', help='Ficheiro de saida')
    parser.add_argument('-r', '--recursive', action='store_true', default=True)
    parser.add_argument('--no-recursive', dest='recursive', action='store_false')
    parser.add_argument('-f', '--format', choices=['csv', 'json'], default='csv')
    parser.add_argument('-w', '--workers', type=int, default=8)
    
    args = parser.parse_args()
    
    output = args.output or f"index_{datetime.now().strftime('%Y%m%d_%H%M%S')}.{args.format}"
    
    indexer = DeterministicIndexer(args.path, args.recursive, args.workers)
    indexer.scan_directory()
    
    if args.format == 'csv':
        indexer.to_csv(output)
    else:
        indexer.to_json(output)
    
    print(f"INDEX: {output}")
    print(f"TOTAL: {len(indexer.results)} ficheiros")

if __name__ == "__main__":
    main()
```

```python
import os
```

```python
def extract_entities(text):
```

```powershell
        Usage = "duphash [-Path <pasta>]"
        Function = {
            param($Params)
            $pasta = if ($Params.Path) { $Params.Path } else { "." }
            $hashGroups = @{}
            Get-ChildItem -Path $pasta -Recurse -File -ErrorAction SilentlyContinue | ForEach-Object {
                $hash = (Get-FileHash $_.FullName -Algorithm MD5 -ErrorAction SilentlyContinue).Hash
                if ($hash) {
                    if (-not $hashGroups.ContainsKey($hash)) { $hashGroups[$hash] = @() }
                    $hashGroups[$hash] += $_
                }
            }
            $hashGroups.GetEnumerator() | Where-Object { $_.Value.Count -gt 1 } |
                ForEach-Object {
                    Write-Host "Hash: $($_.Key)" -ForegroundColor Yellow
                    $_.Value | ForEach-Object { Write-Host "  $($_.Name)" -ForegroundColor Gray }
                }
        }
    }

            Get-ChildItem -Path $pasta -Recurse -File -ErrorAction SilentlyContinue | ForEach-Object {
                $hash = (Get-FileHash $_.FullName -Algorithm MD5 -ErrorAction SilentlyContinue).Hash
                if ($hash -and -not $seen.ContainsKey($hash)) {
                    $seen[$hash] = $true
                } else {
                    Remove-Item -Path $_.FullName -Force -ErrorAction SilentlyContinue
                    Write-Host "Removido: $($_.Name)" -ForegroundColor Red
                }
            }
        }
```

```python
#!/usr/bin/env python3
# indexer.py - Indexador deterministico de ficheiros

import os
import sys
import json
import hashlib
import argparse
from pathlib import Path
from datetime import datetime
from concurrent.futures import ThreadPoolExecutor, as_completed

class DeterministicIndexer:
    def __init__(self, root_path, recursive=True, max_workers=8):
        self.root_path = Path(root_path)
        self.recursive = recursive
        self.max_workers = max_workers
        self.results = []
    
    def hash_file(self, file_path):
        try:
            sha256 = hashlib.sha256()
            with open(file_path, 'rb') as f:
                for chunk in iter(lambda: f.read(65536), b''):
                    sha256.update(chunk)
            return sha256.hexdigest()[:8]
        except:
            return "NAO_DISPONIVEL"
    
    def scan_file(self, file_path):
        try:
            stat = file_path.stat()
            return {
                'hash': self.hash_file(file_path),
                'path': str(file_path),
                'name': file_path.name,
                'size': stat.st_size,
                'extension': file_path.suffix,
                'modified': datetime.fromtimestamp(stat.st_mtime).isoformat(),
                'parent': str(file_path.parent)
            }
        except:
            return None
    
    def scan_directory(self):
        pattern = '**/*' if self.recursive else '*'
        files = list(self.root_path.glob(pattern))
        files = [f for f in files if f.is_file()]
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {executor.submit(self.scan_file, f): f for f in files}
            for future in as_completed(futures):
                result = future.result()
                if result:
                    self.results.append(result)
        return self.results
    
    def to_csv(self, output_path):
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write("HASH|CAMINHO|NOME|TAMANHO|EXTENSAO|ULTIMA_MODIFICACAO|DIRETORIO\n")
            for r in self.results:
                f.write(f"{r['hash']}|{r['path']}|{r['name']}|{r['size']}|{r['extension']}|{r['modified']}|{r['parent']}\n")
    
    def to_json(self, output_path):
        with open(output_path, 'w', encoding='utf-8') as f:
            json.dump({
                'scan_time': datetime.now().isoformat(),
                'path': str(self.root_path),
                'total_files': len(self.results),
                'files': self.results
            }, f, ensure_ascii=False, indent=2)

def main():
    parser = argparse.ArgumentParser(description='Indexador deterministico')
    parser.add_argument('path', help='Caminho para indexar')
    parser.add_argument('-o', '--output', help='Ficheiro de saida')
    parser.add_argument('-r', '--recursive', action='store_true', default=True)
    parser.add_argument('-f', '--format', choices=['csv', 'json'], default='csv')
    parser.add_argument('-w', '--workers', type=int, default=8)
    
    args = parser.parse_args()
    
    output = args.output or f"index_{datetime.now().strftime('%Y%m%d_%H%M%S')}.{args.format}"
    
    indexer = DeterministicIndexer(args.path, args.recursive, args.workers)
    indexer.scan_directory()
    
    if args.format == 'csv':
        indexer.to_csv(output)
    else:
        indexer.to_json(output)
    
    print(f"INDEX: {output}")
    print(f"TOTAL: {len(indexer.results)} ficheiros")

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
# scanner.py - Scanner deterministico de ficheiros

import os
import sys
import json
import hashlib
import argparse
from pathlib import Path
from datetime import datetime
from concurrent.futures import ThreadPoolExecutor

class DeterministicScanner:
    def __init__(self, root_path, max_workers=8):
        self.root_path = Path(root_path)
        self.max_workers = max_workers
    
    def scan_file(self, file_path):
        try:
            stat = file_path.stat()
            content = None
            
            text_extensions = {'.txt', '.md', '.json', '.yaml', '.yml', '.py', '.js', '.html', '.css', '.xml', '.csv', '.log', '.sh', '.ps1'}
            if file_path.suffix.lower() in text_extensions:
                try:
                    with open(file_path, 'r', encoding='utf-8') as f:
                        content = f.read()
                except:
                    content = None
            
            return {
                'path': str(file_path),
                'name': file_path.name,
                'size': stat.st_size,
                'extension': file_path.suffix,
                'modified': datetime.fromtimestamp(stat.st_mtime).isoformat(),
                'hash': hashlib.sha256(open(file_path, 'rb').read()).hexdigest()[:16] if stat.st_size > 0 else '',
                'content_preview': content[:500] if content else None,
                'content_length': len(content) if content else 0
            }
        except:
            return None
    
    def scan(self, recursive=True):
        pattern = '**/*' if recursive else '*'
        files = list(self.root_path.glob(pattern))
        files = [f for f in files if f.is_file()]
        
        results = []
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            for result in executor.map(self.scan_file, files):
                if result:
                    results.append(result)
        return results

def main():
    parser = argparse.ArgumentParser(description='Scanner deterministico')
    parser.add_argument('path', nargs='?', default='.', help='Caminho para scan')
    parser.add_argument('-o', '--output', help='Ficheiro de saida')
    parser.add_argument('-r', '--recursive', action='store_true', default=True)
    parser.add_argument('-w', '--workers', type=int, default=8)
    
    args = parser.parse_args()
    
    scanner = DeterministicScanner(args.path, args.workers)
    results = scanner.scan(args.recursive)
    
    output = args.output or f"scan_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
    
    with open(output, 'w', encoding='utf-8') as f:
        json.dump({
            'scan_time': datetime.now().isoformat(),
            'path': str(args.path),
            'total_files': len(results),
            'files': results
        }, f, ensure_ascii=False, indent=2)
    
    print(f"SCAN: {output}")
    print(f"TOTAL: {len(results)} ficheiros")

if __name__ == "__main__":
    main()
```

```powershell
function dedup-list {

        Group-Object Name |
        Where-Object { $_.Count -gt 1 } |
        ForEach-Object {

            $_.Group | ForEach-Object { Write-Host "  $($_.FullName)" -ForegroundColor Gray }
        }

function dedup-hash {
    param([string]$Path = ".")
    $files = Get-ChildItem -Path $Path -Recurse -File -ErrorAction SilentlyContinue
    $hashes = @{}
    foreach ($f in $files) {
        $hash = (Get-FileHash $f.FullName -Algorithm MD5 -ErrorAction SilentlyContinue).Hash
        if ($hash) {
            if ($hashes.ContainsKey($hash)) {
                Write-Host "REMOVER: $($f.FullName)" -ForegroundColor Red
                Write-Host "  KEEP: $($hashes[$hash])" -ForegroundColor Green
            } else {
                $hashes[$hash] = $f.FullName
            }
        }
    }
}
Set-Alias -Name dh -Value dedup-hash -Scope Global -Force

function dedup-remove {
    param([string]$Path = ".")
    $files = Get-ChildItem -Path $Path -Recurse -File -ErrorAction SilentlyContinue
    $hashes = @{}
    $removed = 0
    foreach ($f in $files) {
        $hash = (Get-FileHash $f.FullName -Algorithm MD5 -ErrorAction SilentlyContinue).Hash
        if ($hash) {
            if ($hashes.ContainsKey($hash)) {
                Remove-Item -Path $f.FullName -Force -ErrorAction SilentlyContinue
                $removed++
                Write-Host "REMOVED: $($f.Name)" -ForegroundColor Red
            } else {
                $hashes[$hash] = $f.FullName
            }
        }
    }
    Write-Host "Total removidos: $removed" -ForegroundColor Green
}
```
